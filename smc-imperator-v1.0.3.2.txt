    float top    = na

// =============================================================================
// 3. CORE LOGIC FUNCTIONS (LuxAlgo Base)
// =============================================================================
leg(int size) =>
    var int l = 0
    bool newLegHigh = high[size] > ta.highest(size)
    bool newLegLow  = low[size]  < ta.lowest(size)
    if newLegHigh
        l := 0
    else if newLegLow
        l := 1
    l

getCurrentStructure(int size) =>
    int currentLeg = leg(size)
    bool newPivot  = ta.change(currentLeg) != 0
    bool pivotLow  = ta.change(currentLeg) == 1
    bool pivotHigh = ta.change(currentLeg) == -1
    if newPivot
        if pivotLow
            swingLow.lastLevel    := swingLow.currentLevel
            swingLow.currentLevel := low[size]
            swingLow.crossed      := false
            swingLow.barTime      := time[size]
            swingLow.barIndex     := bar_index[size]
            trailing.bottom       := swingLow.currentLevel
            trailing.lastBottomTime := swingLow.barTime
            trailing.barTime      := swingLow.barTime
            trailing.barIndex     := swingLow.barIndex
        else if pivotHigh
            swingHigh.lastLevel    := swingHigh.currentLevel
            swingHigh.currentLevel := high[size]
            swingHigh.crossed      := false
            swingHigh.barTime      := time[size]
            swingHigh.barIndex     := bar_index[size]
            trailing.top           := swingHigh.currentLevel
            trailing.lastTopTime   := swingHigh.barTime
            trailing.barTime       := swingHigh.barTime
            trailing.barIndex      := swingHigh.barIndex

updateTrailingExtremes() =>
    trailing.top            := math.max(high, trailing.top)
    trailing.lastTopTime    := trailing.top == high ? time : trailing.lastTopTime
    trailing.bottom         := math.min(low, trailing.bottom)
    trailing.lastBottomTime := trailing.bottom == low ? time : trailing.lastBottomTime

// =============================================================================
// 4. SWING HIGH/LOW DETECTION & DRAWING
// =============================================================================
getCurrentStructure(swingLength)
updateTrailingExtremes()

var line highLine = na
var line lowLine  = na
int rightBarTime = last_bar_time + 20 * (time - time[1])

if na(highLine)
    highLine := line.new(trailing.lastTopTime, trailing.top, rightBarTime, trailing.top, xloc=xloc.bar_time, color=swingHighColor, style=line.style_dashed, width=1)
else
    highLine.set_first_point(chart.point.new(trailing.lastTopTime, na, trailing.top))
    highLine.set_second_point(chart.point.new(rightBarTime, na, trailing.top))

if na(lowLine)
    lowLine := line.new(trailing.lastBottomTime, trailing.bottom, rightBarTime, trailing.bottom, xloc=xloc.bar_time, color=swingLowColor, style=line.style_dashed, width=1)
else
    lowLine.set_first_point(chart.point.new(trailing.lastBottomTime, na, trailing.bottom))
    lowLine.set_second_point(chart.point.new(rightBarTime, na, trailing.bottom))

// =============================================================================
// 5. BREAK OF STRUCTURE (BoS/CHoCH) DETECTION
// =============================================================================

bool crossDown = ta.crossunder(close, swingLow.currentLevel)

if crossUp and not swingHigh.crossed
    swingHigh.crossed := true
    trendBias := 1
    line.new(swingHigh.barTime, swingHigh.currentLevel, time, swingHigh.currentLevel, xloc=xloc.bar_time, color=bullBreakColor, style=line.style_solid, width=1)
    int midIdxHigh = math.round(0.5 * (swingHigh.barIndex + bar_index))
    label.new(midIdxHigh, swingHigh.currentLevel, 'Break', xloc=xloc.bar_index, color=color(na), textcolor=bullBreakColor, style=label.style_label_down, size=size.small)

if crossDown and not swingLow.crossed
    swingLow.crossed := true
    trendBias := -1
    line.new(swingLow.barTime, swingLow.currentLevel, time, swingLow.currentLevel, xloc=xloc.bar_time, color=bearBreakColor, style=line.style_solid, width=1)
    int midIdxLow = math.round(0.5 * (swingLow.barIndex + bar_index))
    label.new(midIdxLow, swingLow.currentLevel, 'Break', xloc=xloc.bar_index, color=color(na), textcolor=bearBreakColor, style=label.style_label_up, size=size.small)

        zn1LblTop.set_text("[UPB] " + str.tostring(z1High, format.mintick))

// 8. FRESHNESS TRACKING LOGIC

// --- Kimi AI freshness reset on BoS (ACTIVE) ---
var bool prevSwingHighCrossed = false
var bool prevSwingLowCrossed  = false
bool newHighBreak = swingHigh.crossed and not prevSwingHighCrossed
bool newLowBreak  = swingLow.crossed  and not prevSwingLowCrossed
bool newStructureFormed = (swingHigh.barTime != prevSwingHighBarTime) or
                          (swingLow.barTime  != prevSwingLowBarTime)  or
                          (trendBias != prevTrendBias) or
                          newHighBreak or newLowBreak
if newStructureFormed
    zn1FormationBar := na
    zn1HasBeenUsed := false
    zn2FormationBar := na
    zn2HasBeenUsed := false
    prevSwingHighBarTime := swingHigh.barTime
    prevSwingLowBarTime  := swingLow.barTime
    prevTrendBias        := trendBias
prevSwingHighCrossed := swingHigh.crossed
prevSwingLowCrossed  := swingLow.crossed

// --- ZN 1 Freshness ---
if na(zn1FormationBar) and not na(z1High)
    zn1FormationBar := bar_index
    zn1HasBeenUsed := false
else if bar_index > zn1FormationBar and not zn1HasBeenUsed and not na(z1High)
    bool zn1UsedByClose   = false
    bool zn1UsedByWickUPB = false
    bool zn1UsedByWickLWB = false
    if trendBias == 1
        zn1UsedByClose   := close <= z1High and close >= z1Low
        zn1UsedByWickUPB := low <= z1High
        zn1UsedByWickLWB := low <= z1Low
    else if trendBias == -1
        zn1UsedByClose   := close >= z1Low and close <= z1High
        zn1UsedByWickUPB := high >= z1High
        zn1UsedByWickLWB := high >= z1Low
    if zn1UsedByClose or zn1UsedByWickUPB or zn1UsedByWickLWB
        zn1HasBeenUsed := true

// --- ZN 2 Freshness ---
if na(zn2FormationBar) and not na(z2High)
    zn2FormationBar := bar_index
    zn2HasBeenUsed := false
else if bar_index > zn2FormationBar and not zn2HasBeenUsed and not na(z2High)
    bool zn2UsedByClose   = false
    bool zn2UsedByWickUPB = false
    bool zn2UsedByWickLWB = false
    if trendBias == 1
        zn2UsedByClose   := close <= z2High and close >= z2Low
        zn2UsedByWickUPB := low <= z2High
        zn2UsedByWickLWB := low <= z2Low
    else if trendBias == -1
        zn2UsedByClose   := close >= z2Low and close <= z2High
        zn2UsedByWickUPB := high >= z2High
        zn2UsedByWickLWB := high >= z2Low
    if zn2UsedByClose or zn2UsedByWickUPB or zn2UsedByWickLWB
        zn2HasBeenUsed := true

// =============================================================================
// 9. TREND FILTER LOGIC (EMA / RSI / STOCH with MTF)
// =============================================================================
// v1.0.3.2: no master switch. A disabled filter simply counts as 'pass'.
// v1.0.4.0: the entry engine MUST hard-gate on trendBlockBuy / trendBlockSell.
string rsiTF   = rsiTimeframe   == '' ? timeframe.period : rsiTimeframe
string stochTF = stochTimeframe == '' ? timeframe.period : stochTimeframe

float emaVal = request.security(syminfo.tickerid, emaTimeframe, ta.ema(close, emaLength))
float rsiVal = request.security(syminfo.tickerid, rsiTF, ta.rsi(close, rsiLength))
float stochK = request.security(syminfo.tickerid, stochTF, ta.sma(ta.stoch(close, high, low, stochK_len), stochSmooth))
float stochD = request.security(syminfo.tickerid, stochTF, ta.sma(ta.sma(ta.stoch(close, high, low, stochK_len), stochSmooth), stochD_len))

// Per-direction filter states
bool emaBuyOk    = not enableEmaFilter   or close > emaVal
bool emaSellOk   = not enableEmaFilter   or close < emaVal
bool rsiBuyOk    = not enableRsiFilter   or rsiVal < rsiOverbought
bool rsiSellOk   = not enableRsiFilter   or rsiVal > rsiOversold
bool stochBuyOk  = not enableStochFilter or stochK < stochOverbought
bool stochSellOk = not enableStochFilter or stochK > stochOversold

bool buyFiltersPass  = emaBuyOk  and rsiBuyOk  and stochBuyOk
bool sellFiltersPass = emaSellOk and rsiSellOk and stochSellOk

bool trendBlockBuy  = not buyFiltersPass
bool trendBlockSell = not sellFiltersPass
bool trendFilterBlock = trendBias == 1 ? trendBlockBuy : trendBias == -1 ? trendBlockSell : false

// Chart plots (EMA on chart; oscillators to status line to avoid scale distortion)
plot(showEmaOnChart ? emaVal : na, 'EMA', color=#FFA000, linewidth=1)
plot(showRsiOnChart ? rsiVal : na, 'RSI', color=#2157f3, display=display.status_line)
plot(showStochOnChart ? stochK : na, 'Stoch %K', color=#9c27b0, display=display.status_line)

// =============================================================================
// 10. SESSION DETECTION (UTC)
// =============================================================================
int utcH = hour(time, 'UTC')
string sessionName = utcH >= 0  and utcH < 7  ? 'Asia' :
                     utcH >= 7  and utcH < 9  ? 'Asia -> Europe' :
                     utcH >= 9  and utcH < 12 ? 'Europe' :
                     utcH >= 12 and utcH < 16 ? 'Europe -> New York' :
                     utcH >= 16 and utcH < 21 ? 'New York' : 'Closed'

// 11. DASHBOARD DISPLAY
// =============================================================================

color c_red    = #F23645
color c_gray   = #878b94
color c_white  = #FFFFFF
color c_yellow = #FFA000
color c_bg_header = color.new(#1E222D, 0)

if showDashboard and barstate.islast
    table.set_position(dashTable, dashPosition == 'Top Right' ? position.top_right : position.top_left)
    table.clear(dashTable, 0, 0, 2, 17)
    int r = 0

    table.cell(dashTable, 0, r, "Zone A1", text_color=c_white, bgcolor=c_bg_header, text_size=size.small)

    table.cell(dashTable, 2, r, "Sell / Short", text_color=c_red, bgcolor=c_bg_header, text_size=size.small)
    r += 1

    table.cell(dashTable, 1, r, not enableEmaFilter ? "Disabled" : emaBuyOk ? "Allow ✓" : "Block ✗", text_color=not enableEmaFilter ? c_gray : emaBuyOk ? c_green : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, not enableEmaFilter ? "Disabled" : emaSellOk ? "Allow ✓" : "Block ✗", text_color=not enableEmaFilter ? c_gray : emaSellOk ? c_green : c_red, text_size=size.small)
    r += 1

    // RSI / Stoch detail rows (optional)
    if showFilterDetails
        table.cell(dashTable, 0, r, "RSI (" + str.tostring(rsiVal, "#.0") + ")", text_color=c_white, text_size=size.small)
        table.cell(dashTable, 1, r, not enableRsiFilter ? "Disabled" : rsiBuyOk ? "Allow ✓" : "Block ✗", text_color=not enableRsiFilter ? c_gray : rsiBuyOk ? c_green : c_red, text_size=size.small)
        table.cell(dashTable, 2, r, not enableRsiFilter ? "Disabled" : rsiSellOk ? "Allow ✓" : "Block ✗", text_color=not enableRsiFilter ? c_gray : rsiSellOk ? c_green : c_red, text_size=size.small)
        r += 1
        table.cell(dashTable, 0, r, "Stoch (" + str.tostring(stochK, "#.0") + ")", text_color=c_white, text_size=size.small)
        table.cell(dashTable, 1, r, not enableStochFilter ? "Disabled" : stochBuyOk ? "Allow ✓" : "Block ✗", text_color=not enableStochFilter ? c_gray : stochBuyOk ? c_green : c_red, text_size=size.small)
        table.cell(dashTable, 2, r, not enableStochFilter ? "Disabled" : stochSellOk ? "Allow ✓" : "Block ✗", text_color=not enableStochFilter ? c_gray : stochSellOk ? c_green : c_red, text_size=size.small)
        r += 1

    // Swing Bias
    table.cell(dashTable, 0, r, "Swing Bias", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? "Bullish ✓" : "Bearish ✗", text_color=trendBias == 1 ? c_green : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? "Bearish ✓" : "Bullish ✗", text_color=trendBias == -1 ? c_red : c_green, text_size=size.small)
    r += 1

    // Zone 1 block
    table.cell(dashTable, 0, r, "--- Zone 1 ---", text_color=c_gray, text_size=size.small)
    table.cell(dashTable, 1, r, "Buy / Long", text_color=c_green, text_size=size.small)
    table.cell(dashTable, 2, r, "Sell / Short", text_color=c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "ZN 1 Formed", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? "OK ✓" : "Invalid", text_color=trendBias == 1 ? c_green : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? "OK ✓" : "Invalid", text_color=trendBias == -1 ? c_green : c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "Waiting Entry for ZN 1", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? (not zn1HasBeenUsed ? "Waiting ✓" : "Done ✗") : "No ✗", text_color=trendBias == 1 ? (not zn1HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? (not zn1HasBeenUsed ? "Waiting ✓" : "Done ✗") : "No ✗", text_color=trendBias == -1 ? (not zn1HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "ZN 1 Validity", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 and not zn1HasBeenUsed ? "Valid" : "Invalid", text_color=trendBias == 1 and not zn1HasBeenUsed ? c_green : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 and not zn1HasBeenUsed ? "Valid" : "Invalid", text_color=trendBias == -1 and not zn1HasBeenUsed ? c_green : c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "ZN 1 Freshness", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? (not zn1HasBeenUsed ? "Fresh" : "Used") : "Invalid", text_color=trendBias == 1 ? (not zn1HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? (not zn1HasBeenUsed ? "Fresh" : "Used") : "Invalid", text_color=trendBias == -1 ? (not zn1HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    r += 1

    // Zone 2 block
    table.cell(dashTable, 0, r, "--- Zone 2 ---", text_color=c_gray, text_size=size.small)
    table.cell(dashTable, 1, r, "Buy / Long", text_color=c_green, text_size=size.small)
    table.cell(dashTable, 2, r, "Sell / Short", text_color=c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "ZN 2 Formed", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? "OK ✓" : "Invalid", text_color=trendBias == 1 ? c_green : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? "OK ✓" : "Invalid", text_color=trendBias == -1 ? c_green : c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "Waiting Entry for ZN 2", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? (not zn2HasBeenUsed ? "Waiting ✓" : "Done ✗") : "No ✗", text_color=trendBias == 1 ? (not zn2HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? (not zn2HasBeenUsed ? "Waiting ✓" : "Done ✗") : "No ✗", text_color=trendBias == -1 ? (not zn2HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "ZN 2 Validity", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 and not zn2HasBeenUsed ? "Valid" : "Invalid", text_color=trendBias == 1 and not zn2HasBeenUsed ? c_green : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 and not zn2HasBeenUsed ? "Valid" : "Invalid", text_color=trendBias == -1 and not zn2HasBeenUsed ? c_green : c_red, text_size=size.small)
    r += 1
    table.cell(dashTable, 0, r, "ZN 2 Freshness", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, trendBias == 1 ? (not zn2HasBeenUsed ? "Fresh" : "Used") : "Invalid", text_color=trendBias == 1 ? (not zn2HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    table.cell(dashTable, 2, r, trendBias == -1 ? (not zn2HasBeenUsed ? "Fresh" : "Used") : "Invalid", text_color=trendBias == -1 ? (not zn2HasBeenUsed ? c_green : c_yellow) : c_red, text_size=size.small)
    r += 1

    // Session row
    table.cell(dashTable, 0, r, "SESSION", text_color=c_white, text_size=size.small)
    table.cell(dashTable, 1, r, sessionName, text_color=c_yellow, text_size=size.small)

    table.cell(dashTable, 0, r, "ACTION", text_color=c_white, bgcolor=c_bg_header, text_size=size.small)
    table.cell(dashTable, 1, r, "STATUS", text_color=c_white, bgcolor=c_bg_header, text_size=size.small)
    table.cell(dashTable, 2, r, "CONDITION", text_color=c_white, bgcolor=c_bg_header, text_size=size.small)
    r += 1

    bool allUsed = zn1HasBeenUsed and zn2HasBeenUsed
    string actionText = trendBias == 1 ? " BUY" : trendBias == -1 ? " SELL" : " WAIT"
    color  actionColor = trendBias == 1 ? c_green : trendBias == -1 ? c_red : c_gray
    string statusText = "WAITING"
    string condText   = "Nulled"

    if trendBias != 0
        if trendFilterBlock
            actionText := "✖ DENIED"
            actionColor := c_red
            statusText := "DENIED"
            condText   := "TREND FILTER BLOCK"
        else if allUsed
            actionText := "⏱ WAIT"
            actionColor := c_yellow
            statusText := "WAIT"
            condText   := "ALL ZONES USED"
        else
            statusText := "READY ENTRY"
            condText   := "RETEST ZN 1 & 2"

    table.cell(dashTable, 0, r, actionText, text_color=c_white, bgcolor=actionColor, text_size=size.small)
    table.cell(dashTable, 1, r, statusText, text_color=trendFilterBlock ? c_red : allUsed ? c_yellow : c_green, text_size=size.small)
    table.cell(dashTable, 2, r, condText, text_color=c_gray, text_size=size.small)
