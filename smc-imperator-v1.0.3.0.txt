type pivot
    float currentLevel = na
    float lastLevel    = na
    bool  crossed      = false
    int   barTime      = time
    int   barIndex     = bar_index

type trailingExtremes
    float top    = na
    float bottom = na
    int   lastTopTime = na
    int   lastBottomTime = na
    int   barTime = na
    int   barIndex = na

var pivot swingHigh           = pivot.new()
var pivot swingLow            = pivot.new()
var trailingExtremes trailing = trailingExtremes.new()
var int trendBias             = 0

// =============================================================================
// 3. CORE LOGIC FUNCTIONS (LuxAlgo Base)
// =============================================================================
leg(int size) =>
    var int l = 0
    bool newLegHigh = high[size] > ta.highest(size)
    bool newLegLow  = low[size]  < ta.lowest(size)
    if newLegHigh
        l := 0
    else if newLegLow
        l := 1
    l

getCurrentStructure(int size) =>
    int currentLeg = leg(size)
    bool newPivot  = ta.change(currentLeg) != 0
    bool pivotLow  = ta.change(currentLeg) == 1
    bool pivotHigh = ta.change(currentLeg) == -1
    if newPivot
        if pivotLow
            swingLow.lastLevel    := swingLow.currentLevel
            swingLow.currentLevel := low[size]
            swingLow.crossed      := false
            swingLow.barTime      := time[size]
            swingLow.barIndex     := bar_index[size]
            trailing.bottom  := swingLow.currentLevel
            trailing.lastBottomTime := swingLow.barTime
            trailing.barTime := swingLow.barTime
            trailing.barIndex:= swingLow.barIndex
        else if pivotHigh
            swingHigh.lastLevel    := swingHigh.currentLevel
            swingHigh.currentLevel := high[size]
            swingHigh.crossed      := false
            swingHigh.barTime      := time[size]
            swingHigh.barIndex     := bar_index[size]
            trailing.top     := swingHigh.currentLevel
            trailing.lastTopTime := swingHigh.barTime
            trailing.barTime := swingHigh.barTime
            trailing.barIndex:= swingHigh.barIndex

updateTrailingExtremes() =>
    trailing.top            := math.max(high, trailing.top)
    trailing.lastTopTime    := trailing.top == high ? time : trailing.lastTopTime
    trailing.bottom         := math.min(low, trailing.bottom)
    trailing.lastBottomTime := trailing.bottom == low ? time : trailing.lastBottomTime

// =============================================================================
// 4. SWING HIGH/LOW DETECTION & DRAWING
// =============================================================================
getCurrentStructure(swingLength)
updateTrailingExtremes()

var line highLine = na
var line lowLine  = na
int rightBarTime = last_bar_time + 20 * (time - time[1])

if na(highLine)
    highLine := line.new(trailing.lastTopTime, trailing.top, rightBarTime, trailing.top, xloc=xloc.bar_time, color=swingHighColor, style=line.style_dashed, width=1)
else
    highLine.set_first_point(chart.point.new(trailing.lastTopTime, na, trailing.top))
    highLine.set_second_point(chart.point.new(rightBarTime, na, trailing.top))

if na(lowLine)
    lowLine := line.new(trailing.lastBottomTime, trailing.bottom, rightBarTime, trailing.bottom, xloc=xloc.bar_time, color=swingLowColor, style=line.style_dashed, width=1)
else
    lowLine.set_first_point(chart.point.new(trailing.lastBottomTime, na, trailing.bottom))
    lowLine.set_second_point(chart.point.new(rightBarTime, na, trailing.bottom))

// =============================================================================
// 5. BREAK OF STRUCTURE (BoS/CHoCH) DETECTION
// =============================================================================
bool crossUp   = ta.crossover(close, swingHigh.currentLevel)
bool crossDown = ta.crossunder(close, swingLow.currentLevel)

if crossUp and not swingHigh.crossed
    swingHigh.crossed := true
    trendBias := 1
    line.new(swingHigh.barTime, swingHigh.currentLevel, time, swingHigh.currentLevel, xloc=xloc.bar_time, color=bullBreakColor, style=line.style_solid, width=1)
    int midIdxHigh = math.round(0.5 * (swingHigh.barIndex + bar_index))
    label.new(midIdxHigh, swingHigh.currentLevel, 'Break', xloc=xloc.bar_index, color=color(na), textcolor=bullBreakColor, style=label.style_label_down, size=size.small)

if crossDown and not swingLow.crossed
    swingLow.crossed := true
    trendBias := -1
    line.new(swingLow.barTime, swingLow.currentLevel, time, swingLow.currentLevel, xloc=xloc.bar_time, color=bearBreakColor, style=line.style_solid, width=1)
    int midIdxLow = math.round(0.5 * (swingLow.barIndex + bar_index))
    label.new(midIdxLow, swingLow.currentLevel, 'Break', xloc=xloc.bar_index, color=color(na), textcolor=bearBreakColor, style=label.style_label_up, size=size.small)

var label zn1Lbl = na

var label zn2Lbl = na
var label zn2LblTop = na
var label zn2LblBot = na

var int   prevSwingLowBarTime   = na

float z1High = na
float z1Low  = na

float z2Low  = na

// =============================================================================
// 7. ZONE CALCULATIONS & DRAWING (ZN 1 & ZN 2)
// =============================================================================
if showZones and not na(trailing.top) and not na(trailing.bottom)
    float priceRange = trailing.top - trailing.bottom
    
    z1High := 0.525 * trailing.top + 0.475 * trailing.bottom
    z1Low  := 0.525 * trailing.bottom + 0.475 * trailing.top
    
    if trendBias == 1
        z2High := trailing.top - 0.618 * priceRange
        z2Low  := trailing.top - 0.786 * priceRange
    else if trendBias == -1
        z2High := trailing.bottom + 0.786 * priceRange
        z2Low  := trailing.bottom + 0.618 * priceRange
    else
        z2High := trailing.top - 0.618 * priceRange
        z2Low  := trailing.top - 0.786 * priceRange
        
    int zn1LeftBarTime = trendBias == 1 ? trailing.lastBottomTime : trendBias == -1 ? trailing.lastTopTime : math.min(trailing.lastTopTime, trailing.lastBottomTime)
    int zn2LeftBarTime = trendBias == 1 ? trailing.lastBottomTime : trendBias == -1 ? trailing.lastTopTime : math.min(trailing.lastTopTime, trailing.lastBottomTime)
    int zn1CenterTime = (zn1LeftBarTime + time) / 2
    int zn2CenterTime = (zn2LeftBarTime + time) / 2
    
    color currentZoneColor = trendBias == 1 ? znBullColor : trendBias == -1 ? znBearColor : znNeutralColor
    string zonePrefix = trendBias == 1 ? "Buy " : trendBias == -1 ? "Sell " : ""
    color labelTextColor = trendBias == 1 ? bullBreakColor : trendBias == -1 ? bearBreakColor : color.white

        zn1LblBot := label.new(rightBarTime, z1Low, "[LWB] " + str.tostring(z1Low, format.mintick), xloc=xloc.bar_time, color=color(na), textcolor=labelTextColor, style=label.style_label_left, size=size.small)

        zn1Box.set_right(rightBarTime)

        zn1Box.set_border_color(color.new(currentZoneColor, 50))

        zn1Lbl.set_text(zonePrefix + 'ZN 1')
        zn1Lbl.set_textcolor(labelTextColor)
        zn1LblTop.set_xy(rightBarTime, z1High)
        zn1LblTop.set_text("[UPB] " + str.tostring(z1High, format.mintick))
        zn1LblTop.set_textcolor(labelTextColor)
        zn1LblBot.set_xy(rightBarTime, z1Low)
        zn1LblBot.set_text("[LWB] " + str.tostring(z1Low, format.mintick))

    if na(zn2Box)

        zn2LblBot := label.new(rightBarTime, z2Low, "[LWB] " + str.tostring(z2Low, format.mintick), xloc=xloc.bar_time, color=color(na), textcolor=labelTextColor, style=label.style_label_left, size=size.small)
    else
        zn2Box.set_left(zn2LeftBarTime)
        zn2Box.set_right(rightBarTime)
        zn2Box.set_top(z2High)
        zn2Box.set_bottom(z2Low)
        zn2Box.set_bgcolor(currentZoneColor)
        zn2Box.set_border_color(color.new(currentZoneColor, 50))
        zn2Lbl.set_xy(zn2CenterTime, math.avg(z2High, z2Low))
        zn2Lbl.set_text(zonePrefix + 'ZN 2')
        zn2Lbl.set_textcolor(labelTextColor)
        zn2LblTop.set_xy(rightBarTime, z2High)
        zn2LblTop.set_text("[UPB] " + str.tostring(z2High, format.mintick))
        zn2LblTop.set_textcolor(labelTextColor)
        zn2LblBot.set_xy(rightBarTime, z2Low)
        zn2LblBot.set_text("[LWB] " + str.tostring(z2Low, format.mintick))
        zn2LblBot.set_textcolor(labelTextColor)

// =============================================================================
// 8. FRESHNESS TRACKING LOGIC
// =============================================================================
var bool prevSwingHighCrossed = false
var bool prevSwingLowCrossed = false
bool newHighBreak = swingHigh.crossed and not prevSwingHighCrossed
bool newLowBreak  = swingLow.crossed and not prevSwingLowCrossed
bool newStructureFormed = (swingHigh.barTime != prevSwingHighBarTime) or
                          (swingLow.barTime != prevSwingLowBarTime) or
                          (trendBias != prevTrendBias) or
                          newHighBreak or newLowBreak

if newStructureFormed
    zn1FormationBar := na
    zn1HasBeenUsed := false
    zn2FormationBar := na
    zn2HasBeenUsed := false
    prevSwingHighBarTime := swingHigh.barTime
    prevSwingLowBarTime := swingLow.barTime
    prevTrendBias := trendBias
    prevSwingHighCrossed := swingHigh.crossed
    prevSwingLowCrossed := swingLow.crossed

if na(zn1FormationBar)
    zn1FormationBar := bar_index
    zn1HasBeenUsed := false
else if bar_index > zn1FormationBar and not zn1HasBeenUsed
    bool zn1Used = false
    if not na(z1High) and not na(z1Low)
        if trendBias == 1
            zn1Used := (close <= z1High and close >= z1Low) or (low <= z1High) or (low <= z1Low)
        else if trendBias == -1
            zn1Used := (close >= z1Low and close <= z1High) or (high >= z1High) or (high >= z1Low)
    if zn1Used
        zn1HasBeenUsed := true

if na(zn2FormationBar)
    zn2FormationBar := bar_index
    zn2HasBeenUsed := false
else if bar_index > zn2FormationBar and not zn2HasBeenUsed
    bool zn2Used = false
    if not na(z2High) and not na(z2Low)
        if trendBias == 1
            zn2Used := (close <= z2High and close >= z2Low) or (low <= z2High) or (low <= z2Low)
        else if trendBias == -1
            zn2Used := (close >= z2Low and close <= z2High) or (high >= z2High) or (high >= z2Low)
    if zn2Used
        zn2HasBeenUsed := true
